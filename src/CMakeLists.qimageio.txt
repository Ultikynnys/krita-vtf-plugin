cmake_minimum_required(VERSION 3.16)
project(krita_vtf_qimageio LANGUAGES CXX)

set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(Qt5 5.15 REQUIRED COMPONENTS Core Gui Widgets Test Concurrent)

add_library(kimg_vtf MODULE
    kis_vtf_codec.cpp
    kis_vtf_qimageio.cpp
)
target_link_libraries(kimg_vtf PRIVATE Qt5::Core Qt5::Gui Qt5::Widgets)
target_include_directories(kimg_vtf PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
set_target_properties(kimg_vtf PROPERTIES PREFIX "")

enable_testing()
add_executable(test_vtf_codec
    ../tests/test_codec.cpp
    kis_vtf_codec.cpp
)
target_link_libraries(test_vtf_codec PRIVATE Qt5::Core Qt5::Gui Qt5::Test)
target_include_directories(test_vtf_codec PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME vtf-codec COMMAND test_vtf_codec)

add_executable(test_vtf_gui_dispatch
    ../tests/test_gui_dispatch.cpp
    vtf_gui_dispatch.cpp
)
target_link_libraries(test_vtf_gui_dispatch PRIVATE Qt5::Core Qt5::Gui Qt5::Widgets Qt5::Test Qt5::Concurrent)
target_include_directories(test_vtf_gui_dispatch PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME vtf-gui-dispatch COMMAND test_vtf_gui_dispatch)
set_tests_properties(vtf-gui-dispatch PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
